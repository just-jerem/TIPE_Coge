\begin{frame}{Cadre thermodynamique}

  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{Conventions}
      \begin{itemize}\small
        \item 1\,kg de fluide (grandeurs massiques)
        \item R\'egime permanent : $\oint \mathrm{d}U = 0$
        \item Fluide : Air
      \end{itemize}

      \eqbox{\Wnet = \Qin - \Qout}
      \vspace{0.5em}
      \eqbox{\eta = \frac{\Wnet}{\Qin} = 1 - \frac{\Qout}{\Qin}}
      \vspace{0.5em}
      \resultbox[Borne de Carnot]{%
        \eta \leq \etaC = 1 - \frac{\Tmin}{\Tmax}}

    \column{0.48\textwidth}
      \begin{block}{Transformations cl\'es --- gaz parfait}
        \small
        \renewcommand{\arraystretch}{1.2}

          Isobare & $\Delta s = C_p\ln(T_f/T_i)$ \\
          Isochore & $\Delta s = C_v\ln(T_f/T_i)$ \\
          Isotherme & $W = rT\ln(v_f/v_i)$ \\
          Isentropique & $T_f/T_i = (P_f/P_i)^{(\gamma-1)/\gamma}$ \\
          \bottomrule
        \end{tabular}
      \end{block}

      \vspace{0.55em}
      \begin{block}{Diagramme T-s}

%% ── Ericsson (transformations + résultat) ───────────────────
\begin{frame}{Cycle d'Ericsson (1/2)}
  \framesubtitle{Transformations et travail net}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{Transformations du cycle}
        \small
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{clc}
          \toprule
          & Transformation & Cste \\
          \midrule
          $1\!\to\!2$ & Compression isotherme      & $\Tmin$ \\
          $2\!\to\!3$ & Chauffage isobare$^*$ & $\Pmax$ \\
          $3\!\to\!4$ & D\'etente isotherme        & $\Tmax$ \\
          $4\!\to\!1$ & Refroidissement isobare$^*$ & $\Pmin$ \\
          \bottomrule
          \multicolumn{3}{l}{\tiny $^*$ via le r\'eg\'en\'erateur}
        \end{tabular}
      \end{block}

    \column{0.46\textwidth}
      \small Sur l'isotherme $T$ (gaz parfait, $Pv=rT$) :\\
      $W = rT\ln(P_i/P_f)$
      \begin{align*}\small
        W_{12} &= -r\Tmin\ln\alpha_p < 0 \\
        W_{34} &= +r\Tmax\ln\alpha_p > 0
      \end{align*}

      \vspace{0.5em}
      \resultbox[Travail net]{%
        \Wnet = r(\Tmax-\Tmin)\ln\alpha_p}
  \end{columns}
\end{frame}

\begin{frame}{Cycle d'Ericsson (2/2)}
  \framesubtitle{Propri\'et\'e cl\'e}
  
  \begin{exampleblock}{Propri\'et\'e cl\'e}
    \normalsize
    $\Wnet \propto \ln\alpha_p$ : croissance monotone.\\[2pt]
    $\Wnet$ \textbf{augmente ind\'efiniment} avec le rapport de pression $\alpha_p=\Pmax/\Pmin$\\[2pt]
    Et cela \textbf{sans modifier le th\'eor\`eme de Carnot} ($\eta = 1 - \Tmin/\Tmax$).\\[2pt]
    $\Rightarrow$ \textbf{S\'eparation totale} entre optimisation de la puissance et du rendement.
  \end{exampleblock}

  \vspace{2em}
  \begin{block}{}

\end{frame}

    \column{0.44\textwidth}
      \begin{exampleblock}{Propri\'et\'e des isochores}
        \normalsize
        Sur un isochore ($dV = 0$) :\\[4pt]
        $\delta W = P\,dV = 0$, \textbf{pas de travail}.\\[6pt]
        $Q_{23} = C_v(\Tmax - \Tmin) > 0$\\[4pt]
        $Q_{41} = C_v(\Tmin - \Tmax) < 0$
      \end{exampleblock}
      
      \vspace{1em}
      \small
      Le r\'eg\'en\'erateur stocke $Q_{23}$ puis le
      restitue lors de $Q_{41}$ :
      aucun \'echange ext\'erieur de chaleur sur ces \'etapes.
  \end{columns}
\end{frame}

\begin{frame}{Cycle de Stirling (2/2)}
  \framesubtitle{Calcul du travail net}

  \begin{block}{Calcul du travail net}
    \normalsize
    Seules les isothermes fournissent du travail.
    (Pour un gaz parfait : $dU = 0$ sur une isotherme) :\\[8pt]
    D\'etente $3\!\to\!4$ \`a $\Tmax$ :\\
    $W_{34} = r\Tmax\ln(\alpha_v) > 0$\\[8pt]
    Compression $1\!\to\!2$ \`a $\Tmin$ :\\
    $W_{12} = r\Tmin\ln(1/\alpha_v) < 0$
  \end{block}

  \vspace{1em}
  \resultbox[Travail net total]{%
    \Wnet = r(\Tmax-\Tmin)\ln\alpha_v}
\end{frame}

  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Double v\'erification syst\'ematique}
        \normalsize
        \textbf{M\'eth. 1} --- entropique :\\[2pt]
        \( \Qin = \Tmax\cdot(s_4-s_3) \)\\[8pt]
        \textbf{M\'eth. 2} --- int\'egration num\'erique $\int P\,dv$\\[2pt]
        Trajectoire d\'ecoup\'ee en 200 points (m\'ethode des trap\`ezes) :
        \[
          W = \sum_i P_i\,\Delta v_i
        \]
      \end{block}

    \column{0.48\textwidth}
      \begin{alertblock}{Crit\`ere d'acceptation num\'erique}
        \normalsize
        Pour valider la consistance de notre int\'egration pas-\`a-pas avec le bilan macroscopique :
        \[
          \frac{|W_{\int P\,dv} - W_{T\Delta s}|}{|W_{T\Delta s}|}
          < 1\,\%
        \]
        \vspace{1em}
        + D\'eductions physiques v\'erifi\'ees :\\[4pt]
        $\eta \in [0,1]$ et $\eta \leq \etaC$
      \end{alertblock}
  \end{columns}

% ── A1 : Relation de Gibbs ───────────────────────────────────
\begin{frame}{A1 --- La relation de Gibbs (1/2)}
  \framesubtitle{Outil fondamental pour toutes les d\'emonstrations : Origine}

  \begin{block}{Origine}
    \normalsize
    Combinaison du 1\textsuperscript{er} et du 2\textsuperscript{e} principe
    pour une transformation \textbf{r\'eversible} :
    \[
      \delta Q_{\text{r\'ev}} = T\,ds
      \quad\text{et}\quad
      \delta W_{\text{r\'ev}} = -P\,dv
    \]
    1\textsuperscript{er} principe : $dU = \delta Q + \delta W = T\,ds - P\,dv$

    Comme $H = U + Pv \Rightarrow dH = dU + P\,dv + v\,dP$ :
    \[
      \boxed{T\,ds = dU + P\,dv = dH - v\,dP}
    \]
    C'est la \textbf{relation de Gibbs}, valable pour tout \'etat
    d'\'equilibre d'un corps pur, m\^eme irr\'eversiblement atteint
    (car $s$, $U$, $H$ sont des fonctions d'\'etat).
  \end{block}
\end{frame}

\begin{frame}{A1 --- La relation de Gibbs (2/2)}
  \framesubtitle{Applications imm\'ediates pour la simulation}

  \begin{exampleblock}{D\'eductions directes thermodynamiques}
    \normalsize
    \textbf{Isobare} ($dP=0$) :\\[2pt]
    $T\,ds = dH = C_p\,dT \;\Rightarrow\; ds = (C_p/T)\,dT$\\[8pt]
    \textbf{Isochore} ($dv=0$) :\\[2pt]
    $T\,ds = dU = C_v\,dT \;\Rightarrow\; ds = (C_v/T)\,dT$\\[8pt]
    \textbf{Isentropique} ($ds=0$) :\\[2pt]
    $C_p\,dT = v\,dP = (rT/P)\,dP$ (mod\`ele gaz parfait)\\[8pt]
    \textbf{Isotherme} ($dT=0$, gaz parfait) :\\[2pt]
    $dU = 0 \Rightarrow \delta Q = \delta W = P\,dv$
  \end{exampleblock}
\end{frame}

% ── A2 : Équations HEOS de CoolProp ─────────────────────────
\begin{frame}{A2 --- \'Equations de Helmholtz (CoolProp)}
  \framesubtitle{Pourquoi et comment CoolProp d\'epasse le gaz parfait}

  \begin{block}{\'Energie libre de Helmholtz r\'eduite}
    \small
    L'\'etat d'un fluide pur est d\'ecrit par l'\'energie de Helmholtz
    r\'eduite $\alpha$, fonction de deux variables r\'eduites :
    \[
      \alpha(\delta, \tau)
      = \frac{A(T,\rho)}{RT}
      = \alpha^o(\delta,\tau) + \alpha^r(\delta,\tau)
    \]
    \vspace{-6pt}
    o\`u $\delta = \rho/\rho_c$, $\tau = T_c/T$,
    $\alpha^o$ = contribution id\'eale (gaz parfait),
    $\alpha^r$ = termes r\'esiduels (interactions mol\'eculaires).
  \end{block}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{Propri\'et\'es d\'eriv\'ees}
        \small
        \vspace{-2pt}
        \[
          \frac{P}{\rho RT}
          = 1 + \delta\!\left(\frac{\partial\alpha^r}{\partial\delta}\right)_{\!\tau}
        \]
        \vspace{-8pt}
        \[
          \frac{H}{RT}
          = 1 + \tau\!\left(\frac{\partial\alpha}{\partial\tau}\right)_{\!\delta}
            + \delta\!\left(\frac{\partial\alpha^r}{\partial\delta}\right)_{\!\tau}
        \]
        \vspace{-8pt}
        \[
          \frac{S}{R}
          = \tau\!\left(\frac{\partial\alpha}{\partial\tau}\right)_{\!\delta}
            - \alpha
        \]
      \end{block}

    \column{0.46\textwidth}
      \begin{exampleblock}{Pour l'air}
        \small
        Les coefficients de $\alpha^r$ sont ajust\'es
        sur des donn\'ees exp\'erimentales couvrant
        $60$--$2000$\,K et jusqu'\`a $2000$\,bar.\\[4pt]
        Pr\'ecision obtenue : $< 0{,}1\,\%$ sur
        $h$, $s$ et $\rho$ pour les conditions
        d'int\'er\^et de cette \'etude.\\[4pt]
        Gaz parfait : $\alpha^r = 0$,
        tous les termes r\'esiduels disparaissent.
      \end{exampleblock}
  \end{columns}

\end{frame}

% ── A3 : Relation isentropique ───────────────────────────────
\begin{frame}{A3 --- Relation isentropique (Brayton)}

  \textbf{\'Enonc\'e :} Montrer que
  $T_2/T_1 = \rp^{(\gamma-1)/\gamma}$
  pour une compression adiabatique r\'eversible.

  \begin{block}{D\'emonstration condens\'ee}
    \small
    Relation de Gibbs en isentropique ($ds=0$) avec l'\'equation du gaz parfait ($Pv = rT$) :
    \[
      T\,ds = dH - v\,dP = C_p\,dT - \frac{rT}{P}\,dP = 0 
      \quad\Rightarrow\quad 
      \frac{dT}{T} = \frac{r}{C_p}\frac{dP}{P}
    \]
    \vspace{-4pt}
    On sait que $r = C_p - C_v$ d'apr\`es la relation de Mayer, donc :
    \[
      \frac{r}{C_p} = 1 - \frac{C_v}{C_p} = 1 - \frac{1}{\gamma} = \frac{\gamma-1}{\gamma}
    \]
    \vspace{-4pt}
    En int\'egrant de $1$ \`a $2$ ($\int dx/x = \ln x$) :
    \[
      \ln\!\left(\frac{T_2}{T_1}\right) = \frac{\gamma-1}{\gamma}\ln\!\left(\frac{P_2}{P_1}\right)
      \quad\Rightarrow\quad
      \boxed{\frac{T_2}{T_1} = \rp^{\,\frac{\gamma-1}{\gamma}}}
    \]
  \end{block}
\end{frame}

% ── A4 : Rendement Brayton ────────────────────────────────────
\begin{frame}{A4 --- D\'emonstration de $\eta_{\text{Brayton}}$}

  \begin{block}{D\'emonstration compl\`ete}
    Poser $x = \rp^{(\gamma-1)/\gamma}$ :
    $T_2 = T_1 x$, $T_4 = T_3/x$.
    \[
      \eta = 1 - \frac{T_4 - T_1}{T_3 - T_2}
           = 1 - \frac{T_3/x - T_1}{T_3 - xT_1}
    \]
    Factoriser par $1/x$ au num\'erateur :
    \[
      = 1 - \frac{(T_3 - xT_1)/x}{T_3 - xT_1}
      = 1 - \frac{1}{x}
    \]
    \[
      \boxed{\eta_{\text{Brayton}} = 1 - \rp^{\,(1-\gamma)/\gamma}}
      \qquad
    \]
    \textbf{Propri\'et\'e :} le facteur $(T_3 - xT_1)$ se simplifie
    exactement --- $\eta$ est ind\'ependant de $T_3$ (temp\'erature
    en sortie de chambre de combustion).
  \end{block}

\end{frame}

% ── A5 : rp* Brayton ─────────────────────────────────────────
\begin{frame}{A5 --- Rapport de pression optimal $\rp^*$ (Brayton)}

  \begin{block}{Maximisation de $\Wnet$}
    \small
    En posant $x = \rp^{(\gamma-1)/\gamma}$ et
    $\tau = \frac{T_3}{T_1}$ (rapport des temp\'eratures source/puits) :
    \[ \Wnet = C_p T_1\!\left(\tau - \frac{\tau}{x} - x + 1\right) \]
    
    Condition n\'ecessaire d'extremum (d\'eriv\'ee premi\`ere nulle) :
    \[ \frac{d\Wnet}{dx} = C_p T_1\!\left(\frac{\tau}{x^2} - 1\right) = 0 \;\Rightarrow\; x = \sqrt{\tau} \]
    
    Condition suffisante (d\'eriv\'ee seconde n\'egative) :
    $d^2\Wnet/dx^2 = -2C_p T_1\tau/x^3 < 0$, c'est bien un \textbf{maximum}.\\
    On trouve en isolant $\rp$ :
    \[ \boxed{\rp^* = \left(\tau\right)^{\!\!\frac{\gamma}{2(\gamma-1)}}} \]
    Application : $T_3/T_1 = 1100/300 \approx 3{,}67$,
    $\gamma = 1{,}4$ $\Rightarrow$ $\rp^* \approx 8{,}6$.
  \end{block}
\end{frame}

% ── A6 : Variation d'entropie isobare ────────────────────────
\begin{frame}{A6 --- Entropie isobare (Ericsson)}

  \textbf{\'Enonc\'e :} Calculer $\Delta s_{23}$ sur l'isobare 2$\to$3
  de $\Tmin$ \`a $\Tmax$ et montrer que $|\Delta s_{23}| = |\Delta s_{41}|$.

  \begin{block}{D\'emonstration}
    \small
    Relation de Gibbs + isobare ($dP=0$) :
    \[
      T\,ds = dH = C_p\,dT
      \;\Rightarrow\;
      ds = \frac{C_p}{T}\,dT
    \]
    Int\'egration de $\Tmin$ \`a $\Tmax$ :
    \[
      \Delta s_{23} = C_p\ln\!\frac{\Tmax}{\Tmin} > 0
    \]
    Pour l'isobare 4$\to$1 (de $\Tmax$ \`a $\Tmin$) :
    \[
      \Delta s_{41} = C_p\ln\!\frac{\Tmin}{\Tmax}
      = -\Delta s_{23}
    \]
    \vspace{-6pt}
    \[
      \Rightarrow |Q_{23}| = C_p(\Tmax-\Tmin) = |Q_{41}|
      \qquad
    \]
    \vspace{-4pt}
    \small\textit{Avec $C_p(T)$ variable (CoolProp) : m\^eme
    conclusion, les deux isobares couvrant $[\Tmin,\Tmax]$.}
  \end{block}

\end{frame}

% ── A7 : Travail net Ericsson ─────────────────────────────────
\begin{frame}{A7 --- Travail net Ericsson}

  \begin{block}{D\'emonstration}
    \textbf{D\'etente 3$\to$4} \`a $\Tmax$, de $\Pmax$ \`a $\Pmin$
    (gaz parfait : $Pv = r\Tmax$) :
    \[
      W_{34} = \int_{v_3}^{v_4} P\,dv
             = r\Tmax\ln\!\frac{v_4}{v_3}
             = r\Tmax\ln\!\frac{P_3}{P_4}
             = r\Tmax\ln\rp
    \]
    \textbf{Compression 1$\to$2} \`a $\Tmin$, de $\Pmin$ \`a $\Pmax$ :
    \[
      W_{12} = r\Tmin\ln\!\frac{v_2}{v_1}
             = r\Tmin\ln\!\frac{\Pmin}{\Pmax}
             = -r\Tmin\ln\rp
    \]
    Les isobares 2$\to$3 et 4$\to$1 sont internes (r\'eg\'en\'eration) :
    $W_{23}$ et $W_{41}$ se compensent globalement.
    \[
      \boxed{\Wnet = r(\Tmax-\Tmin)\ln\rp}
      \qquad
    \]
  \end{block}

\end{frame}

% ── A8 : Démonstration η_Ericsson = η_Carnot (complète) ──────
\begin{frame}{A8 --- $\eta_{\text{Ericsson}} = \etaC$}

  \begin{block}{\'Etape 1 --- R\'eg\'en\'eration exacte \textit{(cf. A6)}}
    \small
    $\Delta s_{23} = -\Delta s_{41}$
    $\Rightarrow$ $|Q_{23}| = |Q_{41}|$ :
    avec $\varepsilon = 1$, les isobares ne font
    \textbf{aucun \'echange ext\'erieur}.
  \end{block}

  \begin{block}{\'Etape 2 --- Bilan entropique du cycle}
    \small
    Seuls \'echanges ext\'erieurs :
    $\Qin = \Tmax(s_4-s_3)$, $\Qout = \Tmin(s_1-s_2)$.

    $\oint ds = 0$ et \'etape 1 $\Rightarrow$
    $(s_3-s_2) = -(s_1-s_4)$, donc :
    \[
      (s_2-s_1) + (s_4-s_3) = 0
      \;\Rightarrow\;
      |s_4-s_3| = |s_2-s_1| \triangleq |\Delta s|
    \]
  \end{block}

  \begin{block}{\'Etape 3 --- Rendement}
    \[
      \boxed{\eta_{\text{Ericsson}}
             = 1 - \frac{\Qout}{\Qin}
             = 1 - \frac{\Tmin\,|\Delta s|}{\Tmax\,|\Delta s|}
             = 1 - \frac{\Tmin}{\Tmax}
             = \etaC}
      \quad
    \]
  \end{block}

\end{frame}

% ── A9 : Démonstration η_Stirling = η_Carnot ────────────────
\begin{frame}{A9 --- D\'emonstration : $\eta_{\text{Stirling}} = \etaC$}

  \begin{block}{\'Etape 1 --- R\'eg\'en\'eration isochore}
    \small
    Sur un isochore ($dv = 0$, $\delta W = 0$) : $\delta Q = C_v\,dT$.
    \textbf{2$\to$3} : $Q_{23} = C_v(\Tmax-\Tmin) > 0$\quad
    \textbf{4$\to$1} : $|Q_{41}| = C_v(\Tmax-\Tmin) = Q_{23}$
    Avec $\varepsilon = 1$ : aucun \'echange ext\'erieur.
  \end{block}

  \begin{block}{\'Etape 2 --- Variations d'entropie des isochores}
    \[
      \Delta s_{23} = C_v\ln\!\frac{\Tmax}{\Tmin}, \quad
      \Delta s_{41} = C_v\ln\!\frac{\Tmin}{\Tmax} = -\Delta s_{23}
    \]
    M\^eme bilan entropique qu'Ericsson :
    $|s_4-s_3| = |s_2-s_1| \triangleq |\Delta s|$
  \end{block}

  \begin{block}{\'Etape 3 --- Rendement}
    Seuls \'echanges ext\'erieurs : isothermes.
    \[
      \boxed{\eta_{\text{Stirling}}
             = 1 - \frac{\Tmin\,|\Delta s|}{\Tmax\,|\Delta s|}
             = 1 - \frac{\Tmin}{\Tmax}
             = \etaC}
      \quad
    \]
  \end{block}

\end{frame}

% ── A10 : Régénérateur imparfait ─────────────────────────────
\begin{frame}{A10 --- R\'eg\'en\'erateur imparfait ($\varepsilon < 1$)}

  \begin{block}{Mod\`ele}
    \small
    $\varepsilon = Q_{\text{transmis}}/Q_{\text{id\'eal}} \in [0,1]$.
    La fraction $(1-\varepsilon)$ non transmise est fournie
    par la source \textbf{externe} :
    \[
      Q_{\text{in,eff}}
      = Q_{34} + (1-\varepsilon)\,C_p(\Tmax-\Tmin)
    \]
  \end{block}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{alertblock}{Rendement effectif}
        \[
          \eta_\varepsilon
          = \frac{r(\Tmax-\Tmin)\ln\rp}
                 {Q_{34} + (1-\varepsilon)C_p\Delta T}
          < \etaC
        \]
        \small
        $\varepsilon = 1$ : $\eta = \etaC$\\
        $\varepsilon = 0{,}85$ : perte $\approx 5$--10\,pts\\
        $\varepsilon = 0$ : $\eta \ll \etaC$
      \end{alertblock}

    \column{0.46\textwidth}
      \begin{exampleblock}{Interpr\'etation}
        \small
        Le r\'eg\'en\'erateur r\'eduit $\Qin$ sans
        changer $\Wnet$.\\[3pt]
        Sans lui (Brayton) :
        $Q_{\text{in}} = Q_{34} + Q_{23}$,
        d'o\`u l'\'ecart de $\approx 19$\,pts avec
        Ericsson \`a conditions identiques.\\[3pt]
        Param\`etre critique : la mesure de $\varepsilon$
        est l'objectif principal de la partie exp\'erimentale.
      \end{exampleblock}
  \end{columns}

\end{frame}

% ── A11 : Exergie ────────────────────────────────────────────
\begin{frame}{A11 --- Exergie et cog\'en\'eration (1/2)}
  \framesubtitle{D\'efinition formelle thermodynamique}

  \begin{block}{D\'efinition}
    \normalsize
    L'exergie d'une chaleur $Q$ disponible \`a $T$
    dans un environnement \`a $T_0$ est le travail util
    \textbf{maximal} extractible, en imaginant une machine de Carnot r\'eversible plac\'ee entre $T$ et $T_0$ :
    \[
      E_x = Q\!\left(1 - \frac{T_0}{T}\right)
    \]
  \end{block}

  \vspace{1em}
  \begin{exampleblock}{Exemples concrets (ambiante $T_0 = 293$\,K)}
    \normalsize
    Vapeur haute enthalpie $200^\circ\mathrm{C}$ : $E_x \approx 38\,\%\cdot Q$\\
    Eau de chauffage $70^\circ\mathrm{C}$ : $E_x \approx 15\,\%\cdot Q$\\
    Rejet condensateur $47^\circ\mathrm{C}$ : $E_x \approx 8\,\%\cdot Q$
  \end{exampleblock}
\end{frame}

\begin{frame}{A11 --- Exergie et cog\'en\'eration (2/2)}
  \framesubtitle{Outil d'\'evaluation \`a la faveur de la cog\'en\'eration}
  
  \begin{block}{Destruction d'exergie au condenseur pur}
    \normalsize
    Rejeter brutalement $\Qout$ \`a $T_0$ cr\'e\'e irr\'em\'ediablement de l'entropie :
    \[
      \dot S_\text{irr} = \Qout\!\left(\frac{1}{T_0} - \frac{1}{T_\text{cond}}\right) \geq 0
    \]
  \end{block}

  \vspace{0.5em}
  \begin{alertblock}{Motivation thermo-conceptuelle pour la cog\'en\'eration}
    \normalsize
    Dans une centrale classique,
    $\Qout$ est rejet\'e \`a $T_\text{cond} \approx T_0$ :
    son exergie r\'esiduelle est quasi nulle ($\approx 8\,\%$)
    et \textbf{d\'etruite irr\'eversiblement} par transfert thermique hors l'\'equilibre vers le milieu ambiant.\\[6pt]
    En cog\'en\'eration certifi\'ee, $\Qout$ est valoris\'e r\'efl\'echit
    \underline{avant} d'atteindre le point mort et limitant $T_0$ :
    on r\'ecup\`ere d'abord l'essentiel du co\^ut de pompage sous forme d'exergie \textit{r\'eutilisable}.\\[6pt]
    C'est la preuve fondamentale, au sens du 2\textsuperscript{e} principe,
    en faveur d'une telle technologie.
  \end{alertblock}
\end{frame}

% ── A12 : Code Ericsson ───────────────────────────────────────
\begin{frame}[fragile]{A12 --- Code : Ericsson.py}

  \begin{exampleblock}{}
  \begin{lstlisting}
from CoolProp.CoolProp import PropsSI as CP
from functools import lru_cache
import numpy as np

fluid = "Air"

@lru_cache(maxsize=50000)
def CP_cached(out, i1, v1, i2, v2, fl):
    return CP(out, i1, v1, i2, v2, fl)

def get_s(T, P): return CP_cached("S","T",T,"P",P,fluid)
def get_h(T, P): return CP_cached("H","T",T,"P",P,fluid)

def cycle_ericsson(T_min=300, T_max=1100, P_min=2e5, P_max=20e5):
    s1, s2 = get_s(T_min,P_min), get_s(T_min,P_max)
    s3, s4 = get_s(T_max,P_max), get_s(T_max,P_min)

    Q_in  = T_max*(s4-s3)   # chaleur recue  (isotherme 3->4)
    Q_out = T_min*(s1-s2)   # chaleur cedee  (isotherme 1->2)
    W_net = Q_in - Q_out
    eta   = W_net / Q_in

    # Validation croisee int(P dv) via trapeze sur 200 pts
    # Critere : |W_pdv - W_net| / |W_net| < 1 %
    return {"Q_in":Q_in, "Q_out":Q_out, "W_net":W_net, "eta":eta}

def cycle_brayton(T1=300, P1=2e5, pressure_ratio=10, T3=1100):
    P2 = P1 * pressure_ratio

    # Etat 2 : compression isentropique (s constant = s1)
    s1 = get_s(T1, P1)
    T2 = CP("T","P",P2,"S",s1,fluid)  # T tel que s(T,P2)=s1

    # Etat 3 : fin chauffage isobare
    s3 = get_s(T3, P2)

    # Etat 4 : detente isentropique (s constant = s3)
    T4 = CP("T","P",P1,"S",s3,fluid)

    h1,h2 = get_h(T1,P1), get_h(T2,P2)
    h3,h4 = get_h(T3,P2), get_h(T4,P1)

    W_comp = h2-h1          # travail compresseur (recu)
    W_turb = h3-h4          # travail turbine (fourni)
    W_net  = W_turb - W_comp
    Q_in   = h3-h2

    return {"W_net":W_net, "Q_in":Q_in, "eta":W_net/Q_in,
            "T2":T2, "T4":T4}

def cycle_stirling(T_min=300, T_max=1100, V_min=0.3, V_max=1.0):
    # Pression via equation d'etat : rho = 1/v
    def get_P(T,V):
        return CP_cached("P","T",T,"D",1/V,fluid)

    P1 = get_P(T_min, V_max)  # etat 1 : froid, grand vol.
    P2 = get_P(T_min, V_min)  # etat 2 : froid, petit vol.
    P3 = get_P(T_max, V_min)  # etat 3 : chaud, petit vol.
    P4 = get_P(T_max, V_max)  # etat 4 : chaud, grand vol.

    s1,s2 = get_s(T_min,P1), get_s(T_min,P2)
    s3,s4 = get_s(T_max,P3), get_s(T_max,P4)

    Q_in  = T_max*(s4-s3)   # isotherme 3->4
    Q_out = T_min*(s1-s2)   # isotherme 1->2

    # Chaleur isochores (regeneration interne)
    Q_regen = get_h(T_max,P3) - get_h(T_min,P2)

    W_net = Q_in - Q_out
    return {"Q_in":Q_in, "Q_out":Q_out, "W_net":W_net,
            "Q_regen":Q_regen, "eta":W_net/Q_in,
            "eta_carnot":1-T_min/T_max}
